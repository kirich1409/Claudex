# Claudex UI Kit Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the Claudex UI design system in `commonMain` — theme tokens, typography, spacing, and 9 reusable Compose components that wrap Material3, enforcing the Claudex visual language.

**Architecture:** `ClaudexTheme` wraps M3's `MaterialTheme` with a hand-crafted `ColorScheme` (no dynamic color), a custom `Typography` using bundled Inter + Roboto Mono fonts, and a `CompositionLocal`-based spacing scale. Every component references `MaterialTheme.colorScheme.*` only — no raw `Color` literals outside `ClaudexColors.kt`.

**Tech Stack:** Compose Multiplatform 1.10.0, Material3 1.10.0-alpha05 (bundled), Kotlin 2.3.0, `runComposeUiTest` (CMP desktop) for component tests, `kotlin.test` for pure unit tests.

```kotlin
// commonMain/kotlin/dev/androidbroadcast/claudex/ui/theme/ClaudexTheme.kt
package dev.androidbroadcast.claudex.ui.theme

import androidx.compose.foundation.isSystemInDarkTheme
import androidx.compose.material3.MaterialTheme
import androidx.compose.runtime.Composable
import androidx.compose.runtime.CompositionLocalProvider

@Composable
public fun ClaudexTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    content: @Composable () -> Unit,
) {
    val colorScheme = if (darkTheme) ClaudexColors.darkScheme else ClaudexColors.lightScheme
    CompositionLocalProvider(
        LocalSpacing provides ClaudexSpacing(),
        LocalCodeFont provides RobotoMono,
    ) {
        MaterialTheme(
            colorScheme = colorScheme,
            typography = ClaudexTypography,
            content = content,
        )
    }
}
```

```kotlin
// commonMain/kotlin/dev/androidbroadcast/claudex/ui/component/button/ClaudexButton.kt
package dev.androidbroadcast.claudex.ui.component.button

import androidx.compose.foundation.layout.size
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.automirrored.filled.Send
import androidx.compose.material3.*
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.vector.ImageVector
import androidx.compose.ui.unit.dp

public enum class ButtonVariant { Primary, Ghost, Destructive }

@Composable
public fun ClaudexButton(
    label: String,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    variant: ButtonVariant = ButtonVariant.Primary,
    enabled: Boolean = true,
    leadingIcon: ImageVector? = null,
) {
    when (variant) {
        ButtonVariant.Primary -> Button(
            onClick = onClick,
            modifier = modifier,
            enabled = enabled,
        ) {
            if (leadingIcon != null) {
                Icon(leadingIcon, contentDescription = null, modifier = Modifier.size(18.dp))
            }
            Text(label)
        }

        ButtonVariant.Ghost -> OutlinedButton(
            onClick = onClick,
            modifier = modifier,
            enabled = enabled,
        ) {
            if (leadingIcon != null) {
                Icon(leadingIcon, contentDescription = null, modifier = Modifier.size(18.dp))
            }
            Text(label)
        }

        ButtonVariant.Destructive -> Button(
            onClick = onClick,
            modifier = modifier,
            enabled = enabled,
            colors = ButtonDefaults.buttonColors(
                containerColor = MaterialTheme.colorScheme.error,
                contentColor = MaterialTheme.colorScheme.onError,
            ),
        ) {
            if (leadingIcon != null) {
                Icon(leadingIcon, contentDescription = null, modifier = Modifier.size(18.dp))
            }
            Text(label)
        }
    }
}

@Composable
public fun ClaudexIconButton(
    contentDescription: String,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    icon: ImageVector = Icons.AutoMirrored.Filled.Send,
    enabled: Boolean = true,
) {
    IconButton(onClick = onClick, modifier = modifier, enabled = enabled) {
        Icon(icon, contentDescription = contentDescription)
    }
}
```

```kotlin
    } else {
        FilterChip(
            selected = false,
            onClick = onClick,
```

/**
 * Renders assistant markdown text with basic inline styling:
 * **bold**, `code`, and fenced code blocks (``` ... ```).
 * Block-level features (headers, lists) are rendered as plain text for MVP.
 */
@Composable
public fun AssistantMessage(
    text: String,
    modifier: Modifier = Modifier,
) {
    val spacing = LocalSpacing.current
    val codeFont = LocalCodeFont.current
    val codeBg = MaterialTheme.colorScheme.surfaceContainerHigh

    // Split into fenced code blocks vs inline text
    val parts = remember(text) { splitCodeBlocks(text) }

    Column(modifier = modifier.fillMaxWidth()) {
        for (part in parts) {
            if (part.isCodeBlock) {
                Card(
                    shape = RoundedCornerShape(8.dp),
                    colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surfaceContainerHighest),
                    modifier = Modifier.fillMaxWidth().padding(vertical = spacing.xs),
                ) {
                    Text(
                        text = part.content,
                        fontFamily = codeFont,
                        style = MaterialTheme.typography.bodyMedium.copy(fontFamily = codeFont),
                        modifier = Modifier
                            .horizontalScroll(rememberScrollState())
                            .padding(spacing.sm),
                    )
                }
            } else {
                val annotated = remember(part.content) {
                    buildAnnotatedString {
                        val inline = part.content
                        var i = 0
                        while (i < inline.length) {
                            when {
                                inline.startsWith("**", i) -> {
                                    val end = inline.indexOf("**", i + 2)
                                    if (end != -1) {
                                        withStyle(SpanStyle(fontWeight = FontWeight.Bold)) {
                                            append(inline.substring(i + 2, end))
                                        }
                                        i = end + 2
                                    } else { append(inline[i]); i++ }
                                }
                                inline[i] == '`' -> {
                                    val end = inline.indexOf('`', i + 1)
                                    if (end != -1) {
                                        withStyle(SpanStyle(fontFamily = codeFont, background = codeBg)) {
                                            append(inline.substring(i + 1, end))
                                        }
                                        i = end + 1
                                    } else { append(inline[i]); i++ }
                                }
                                else -> { append(inline[i]); i++ }
                            }
                        }
                    }
                }
                Text(
                    text = annotated,
                    style = MaterialTheme.typography.bodyLarge,
                    modifier = Modifier.padding(vertical = spacing.xs),
                )
            }
        }
    }
}

internal data class TextPart(val content: String, val isCodeBlock: Boolean)

internal fun splitCodeBlocks(text: String): List<TextPart> {
    val result = mutableListOf<TextPart>()
    val fence = "```"
    var start = 0
    while (true) {
        val fenceStart = text.indexOf(fence, start)
        if (fenceStart == -1) {
            if (start < text.length) result.add(TextPart(text.substring(start), isCodeBlock = false))
            break
        }
        if (fenceStart > start) result.add(TextPart(text.substring(start, fenceStart), isCodeBlock = false))
        val codeStart = text.indexOf('\n', fenceStart) + 1
        val fenceEnd = text.indexOf(fence, codeStart)
        if (fenceEnd == -1) {
            result.add(TextPart(text.substring(fenceStart + 3), isCodeBlock = true))
            break
        }
        result.add(TextPart(text.substring(codeStart, fenceEnd).trimEnd(), isCodeBlock = true))
        start = fenceEnd + 3
    }
    return result
}

```kotlin
@Composable
public fun ToolUseCard(
    toolName: String,
    jsonBody: String,
    modifier: Modifier = Modifier,
    elapsedLabel: String? = null,
) {
    val spacing = LocalSpacing.current
    val codeFont = LocalCodeFont.current
    var expanded by remember { mutableStateOf(false) }

    Card(
        shape = RoundedCornerShape(8.dp),
        colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surfaceContainer),
        modifier = modifier.fillMaxWidth(),
    ) {
        Column(Modifier.padding(spacing.sm)) {
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier.fillMaxWidth(),
            ) {
                Icon(
                    imageVector = Icons.Default.Build,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.size(16.dp),
                )
                Spacer(Modifier.width(spacing.xs))
                Text(
                    text = toolName,
                    style = MaterialTheme.typography.labelSmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.weight(1f),
                )
                if (elapsedLabel != null) {
                    Text(
                        text = elapsedLabel,
                        style = MaterialTheme.typography.labelSmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                    Spacer(Modifier.width(spacing.xs))
                }
                IconButton(
                    onClick = { expanded = !expanded },
                    modifier = Modifier.size(24.dp),
                ) {
                    Icon(
                        imageVector = if (expanded) Icons.Default.ExpandLess else Icons.Default.ExpandMore,
                        contentDescription = if (expanded) "Collapse tool details" else "Expand tool details",
                        modifier = Modifier.size(16.dp),
                    )
                }
            }
            AnimatedVisibility(visible = expanded) {
                Text(
                    text = jsonBody,
                    fontFamily = codeFont,
                    style = MaterialTheme.typography.bodyMedium.copy(fontFamily = codeFont),
                    modifier = Modifier
                        .fillMaxWidth()
                        .horizontalScroll(rememberScrollState())
                        .padding(top = spacing.xs),
                )
            }
        }
    }
}
```

```kotlin
// commonMain/kotlin/dev/androidbroadcast/claudex/ui/component/layout/DragHandle.kt
package dev.androidbroadcast.claudex.ui.component.layout

import androidx.compose.foundation.background
import androidx.compose.foundation.hoverable
import androidx.compose.foundation.interaction.MutableInteractionSource
import androidx.compose.foundation.interaction.collectIsHoveredAsState
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.height
import androidx.compose.foundation.layout.width
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.MaterialTheme
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp

/**
 * Visual drag handle for the sidebar-to-chat pane splitter.
 * Wiring to [PaneExpansionState] is done at the layout level (ListDetailPaneScaffold).
 */
@Composable
public fun DragHandle(
    modifier: Modifier = Modifier,
) {
    val interactionSource = remember { MutableInteractionSource() }
    val hovered by interactionSource.collectIsHoveredAsState()
    val color = if (hovered) MaterialTheme.colorScheme.outline
                else MaterialTheme.colorScheme.outlineVariant
    Box(
        modifier = modifier
            .width(4.dp)
            .height(24.dp)
            .background(color = color, shape = RoundedCornerShape(2.dp))
            .hoverable(interactionSource),
    )
}
```
